const express = require('express'); const app = express(); app.use(express.json()); // Nümunə verilənlər bazası (Bütün istifadəçilər) let usersDB = { "7171314206": { id: "7171314206", name: "Orxan Orxan", balance: 189, profit_per_hour: 780, last_update_time: Date.now() }, "1234567890": { id: "1234567890", name: "Elvin Məmmədov", balance: 3450, profit_per_hour: 1200, last_update_time: Date.now() }, "9876543210": { id: "9876543210", name: "Anar Qasımov", balance: 920, profit_per_hour: 450, last_update_time: Date.now() } }; const MAX_LIMIT_SECONDS = 3 * 3600; // 3 saat // 1. Sinxronizasiya və Arxa fon qazancının hesablanması API-si app.post('/api/sync', (req, res) => { const { userId } = req.body; let user = usersDB[userId]; if (!user) { return res.status(404).json({ error: "İstifadəçi tapılmadı" }); } const currentTime = Date.now(); const elapsedSeconds = Math.floor((currentTime - user.last_update_time) / 1000); let earnedCoins = 0; if (elapsedSeconds > 0) { const effectiveSeconds = Math.min(elapsedSeconds, MAX_LIMIT_SECONDS); earnedCoins = Math.floor((user.profit_per_hour / 3600) * effectiveSeconds); if (earnedCoins > 0) { user.balance += earnedCoins; user.last_update_time = currentTime; } } // Oyundakı ümumi qeydiyyatda olan istifadəçi sayı const totalUsersCount = Object.keys(usersDB).length; res.json({ oldBalance: user.balance - earnedCoins, // Əvvəlki balans (gəlməzdən əvvəlki) earnedCoins: earnedCoins, // Arxa fonda yığılan xal balance: user.balance, // Yeni ümumi balans profit_per_hour: user.profit_per_hour, totalUsersCount: totalUsersCount // Qeydiyyatda olanların ümumi sayı }); }); // 2. Liderlər lövhəsi (Xallarına görə sıralanmış siyahı) API-si app.get('/api/leaderboard', (req, res) => { // İstifadəçiləri balanslarına (xallarına) görə çoxdan aza doğru sıralayırıq const sortedUsers = Object.values(usersDB) .sort((a, b) => b.balance - a.balance) .map((u, index) => ({ rank: index + 1, name: u.name, balance: u.balance })); res.json({ totalUsers: sortedUsers.length, leaderboard: sortedUsers }); }); app.listen(3000, () => { console.log('Server işləyir...'); });
